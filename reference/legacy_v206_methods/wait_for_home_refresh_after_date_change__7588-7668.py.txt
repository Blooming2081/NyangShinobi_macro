    def wait_for_home_refresh_after_date_change(self):
        """날짜 변경 확인 후 홈 화면이 안정적으로 표시될 때까지 기다립니다.

        예전처럼 기존 홈 이미지가 반드시 사라졌는지를 요구하지 않습니다.
        전환 애니메이션이 너무 짧으면 사라짐을 놓쳐 작업 스레드가 종료될 수
        있었기 때문에, 짧은 안전 대기 후 홈 화면 연속 감지만 확인합니다.
        """
        self.log("날짜 변경 후 홈 화면 갱신을 기다립니다.")
        self.mark_progress("날짜 변경 후 홈 화면 대기 시작")

        # 서버 날짜와 홈 UI가 갱신될 시간을 먼저 확보합니다.
        for _ in range(30):
            self.checkpoint()
            time.sleep(0.1)

        deadline = time.time() + 60.0
        stable_count = 0
        last_log = 0.0
        next_loading_click_at = time.time()
        while time.time() < deadline:
            self.checkpoint()
            screen = self.adb_screencap()
            followup_found, followup_x, followup_y, followup_score = (
                self.find_nav_image_near(
                    NAV_TEMPLATES["date_change_followup_close"],
                    DATE_CHANGED_FOLLOWUP_CLOSE_CENTER_RATIO,
                    DATE_CHANGED_FOLLOWUP_CLOSE_TOLERANCE_RATIO,
                    threshold=0.82,
                    screen=screen,
                )
            )
            if followup_found:
                self.log(
                    "날짜 변경 후 일일 팝업 닫기 발견 "
                    f"/ 일치율 {followup_score:.3f} → 클릭"
                )
                self.adb_click(followup_x, followup_y)
                self.mark_progress("날짜 변경 후 일일 팝업 닫기")
                stable_count = 0
                time.sleep(0.4)
                continue
            ready = self._home_screen_ready(screen)
            now = time.time()
            if ready:
                stable_count += 1
                if stable_count >= 2:
                    self.log("날짜 변경 후 홈 화면 확인 완료 → 기존 작업을 다시 시작합니다.")
                    self.mark_progress("날짜 변경 후 홈 화면 확인")
                    return
            else:
                stable_count = 0
                # 날짜 갱신 뒤 게임이 TAP TO START 화면으로 돌아오는 경우에만
                # 기존 재시작 루틴과 같은 안전한 중앙 좌표를 눌러 홈까지 진행합니다.
                if screen is not None and now >= next_loading_click_at:
                    height, width = screen.shape[:2]
                    click_x = int(RESTART_SAFE_CLICK_RATIO[0] * width)
                    click_y = int(RESTART_SAFE_CLICK_RATIO[1] * height)
                    self.run_adb(
                        [
                            "shell", "input", "tap",
                            str(click_x), str(click_y),
                        ],
                        timeout=5,
                    )
                    self.log("날짜 변경 후 로딩 화면 넘기기 클릭")
                    next_loading_click_at = now + RESTART_CLICK_INTERVAL

            if now - last_log >= 5.0:
                self.log("날짜 변경 후 홈 화면 확인 중")
                last_log = now
            time.sleep(0.4)

        # 홈 감지가 잠시 실패해도 작업 스레드를 끝내지 않습니다.
        # 이후 진입 함수가 홈 버튼 이미지를 다시 찾으며 재시도하게 합니다.
        self.log("홈 화면 감지 시간이 초과됐지만 작업을 종료하지 않고 재진입을 시도합니다.")
        self.mark_progress("날짜 변경 후 홈 재진입 시도")

    def reset_run_stats(self):
        self.run_started_at = time.time()
        self.run_started_at_text = datetime.now().astimezone().isoformat(timespec="seconds")
        self.last_completed_at = self.run_started_at
