    def _use_mirror_bigi_and_verify(self, initial_center=None, initial_screen=None):
        """발광 얼굴을 누르고 유효한 전투 프레임에서 사라질 때만 완료 처리합니다.

        카드 선택 오버레이는 우측 HUD를 어둡게 만들어 사용 가능한 얼굴도
        미감지로 보입니다. 그 화면을 단순 소실로 누적하면 입력이 씹힌 경우를
        성공으로 오인하므로 카드·결과 전환은 완료 판정에서 제외합니다.
        호출 측이 같은 프레임에서 카드 패널 부재와 발광 얼굴을 확인했다면
        추가 캡처를 기다리지 않고 그 좌표를 즉시 입력합니다.
        """
        face_center = initial_center or BIGI_MIRROR_ALLOWED_CENTER
        best_area = 0
        if initial_center is None:
            face_deadline = time.monotonic() + 2.0
            while time.monotonic() < face_deadline:
                self.checkpoint()
                screen = self._raw_screencap(max_cache_age=0.0)
                present, area, detected_center = self._detect_mirror_bigi_hud_face(
                    screen
                )
                screen = None
                best_area = max(best_area, area)
                if present:
                    face_center = detected_center
                    break
                if self.stop_event.wait(0.05):
                    return False
            else:
                self.log(
                    "거울 비기 카드 선택 후 허용 좌표의 발광 얼굴을 확인하지 "
                    f"못함 → 사용 완료 처리 보류 / 최고 영역 {best_area}px"
                )
                return False

        for attempt in range(3):
            self.checkpoint()
            if attempt == 0 and initial_screen is not None:
                # 메인 루프는 이미 이 프레임에서 확인 버튼·카드 패널 부재와
                # 발광 얼굴을 모두 검사했습니다. 재캡처 2회와 0.12초 대기는
                # 카드 화면으로 넘어갈 틈만 늘리므로 즉시 탭합니다.
                self._adb_click_mirror_reference(*face_center, screen=initial_screen)
            else:
                # 자체 탐색 호출과 재시도에서는 이전 화면을 재사용하지 않고
                # 카드 패널 부재·얼굴 상태를 새 프레임으로 교차 확인합니다.
                before_screen = self._raw_screencap(max_cache_age=0.0)
                if before_screen is None:
                    return False
                before_confirm, _, _, _ = self.find_mirror_skill_confirm(
                    screen=before_screen, threshold=0.62
                )
                before_layout, _, _, _ = self._detect_mirror_card_layout(before_screen)
                if before_confirm or before_layout is not None:
                    self.log("거울 비기 입력 직전 카드 패널 감지 → 비기 좌표 입력 보류")
                    return False
                before_face, _, _ = self._detect_mirror_bigi_hud_face(
                    before_screen
                )
                if not before_face:
                    return False
                if self.stop_event.wait(0.12):
                    return False
                stable_screen = self._raw_screencap(max_cache_age=0.0)
                if stable_screen is None:
                    return False
                stable_confirm, _, _, _ = self.find_mirror_skill_confirm(
                    screen=stable_screen, threshold=0.62
                )
                stable_layout, _, _, _ = self._detect_mirror_card_layout(stable_screen)
                stable_face, _, stable_center = self._detect_mirror_bigi_hud_face(
                    stable_screen
                )
                if stable_confirm or stable_layout is not None or not stable_face:
                    self.log("거울 비기 입력 직전 카드/전투 전환 확인 → 입력 보류")
                    return False
                face_center = stable_center
                self._adb_click_mirror_reference(*face_center, screen=stable_screen)
            absent_frames = 0
            absent_since = None
            for _ in range(12):
                if self.stop_event.wait(0.10):
                    return False
                after_screen = self._raw_screencap(max_cache_age=0.0)
                if after_screen is None or after_screen.size == 0 or float(np.std(after_screen[::8, ::8])) < 8.0:
                    absent_frames = 0
                    absent_since = None
                    continue

                # 실제 저장 화면에서 카드 오버레이가 열린 동안 발광 얼굴은
                # 화면에 남아 있어도 어두워져 얼굴 감지값이 False가 됩니다.
                # 이때는 사용 성공으로 누적하지 않고 즉시 메인 카드 처리로
                # 제어를 돌려 확인 후 다시 비기 입력을 시도합니다.
                card_screen, _, _, card_score = self.find_mirror_skill_confirm(
                    screen=after_screen, threshold=0.62
                )
                card_layout, _, _, _ = self._detect_mirror_card_layout(after_screen)
                if card_screen or card_layout is not None:
                    self.log(
                        "거울 비기 입력 검증 중 카드 선택 화면 전환 감지 "
                        f"/ 확인 {card_score:.3f}, 카드 {card_layout or '미확정'}장 "
                        "→ 사용 완료 처리 취소 후 "
                        "카드 선택을 먼저 진행합니다."
                    )
                    return False

                result_region = after_screen[
                    int(after_screen.shape[0] * 0.18):
                    int(after_screen.shape[0] * 0.33), :
                ]
                mirror_failed, _, _, _ = self.find_battle_image(
                    self.image_path(8), threshold=0.68, screen=result_region
                )
                mirror_cleared, _, _, _ = self.find_battle_image(
                    self.image_path(6), screen=result_region
                )
                if mirror_failed or mirror_cleared:
                    self.log(
                        "거울 비기 입력 검증 중 결과 화면 전환 감지 → "
                        "얼굴 미감지를 사용 완료로 간주하지 않고 결과 처리를 우선합니다."
                    )
                    return False

                present, area, detected_center = (
                    self._detect_mirror_bigi_hud_face(after_screen)
                )
                after_screen = None
                best_area = max(best_area, area)
                if present:
                    face_center = detected_center
                    absent_frames = 0
                    absent_since = None
                else:
                    absent_frames += 1
                    if absent_since is None:
                        absent_since = time.monotonic()
                    if absent_frames >= 5 and time.monotonic() - absent_since >= 1.0:
                        self.log(
                            "거울 비기 사용 완료 확인 → 허용 좌표의 발광 "
                            "얼굴이 1초 이상·5프레임 이상 연속 사라짐"
                        )
                        return True
            if attempt < 2:
                self.log(
                    "거울 비기 얼굴이 입력 후에도 남아 있음 → "
                    f"0.10초 후 재입력 {attempt + 1}/2"
                )
                if self.stop_event.wait(0.10):
                    return False

        self.log(
            "거울 비기 사용 성공을 확인하지 못함 → 검색 종료하지 않음 "
            f"/ 발광 영역 최고 {best_area}px"
        )
        return False
