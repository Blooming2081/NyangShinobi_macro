    def enter_mission_from_home(self):
        self.log("일반 임무 진입 시작")
        # 재진입 오류 직후 BlueStacks Store/Game Center가 전면에 있으면
        # 그 화면에서 홈 임무 버튼을 40초간 찾지 않습니다. 먼저 Unity 게임을
        # 전면 실행하고 홈 준비를 확인한 뒤 기존 임무 진입 절차를 시작합니다.
        foreground = self.detect_foreground_package(include_system=True)
        if (
            foreground is not None
            and foreground not in (GAME_PACKAGE, "com.android.systemui")
        ):
            self.log(
                "임무 재진입 전 BlueStacks Game Center/게임 외 화면 감지 "
                f"({foreground}) → 냥냥 시노비를 먼저 전면 실행합니다."
            )
            self.game_package = GAME_PACKAGE
            if not self.restart_game("임무 재진입 전 Game Center 복구"):
                self.log(
                    "게임 단독 전면 실행 실패 → BlueStacks와 게임 전체 "
                    "복구로 전환합니다."
                )
                self._launch_bluestacks_and_game()
            self.mark_progress("임무 재진입 전 Game Center 복구")

        # 이전 성공/실패 직후 늦게 나타난 레벨업 화면에서도 정상 복귀합니다.
        self.handle_mission_level_up_popup(timeout=0.8)
        screen = self._raw_screencap()
        if self._mission_entry_screen_ready(screen):
            self.log(
                "이미 임무 진입 전 화면 확인 → 홈의 임무 버튼 검색을 생략합니다."
            )
            self.mark_progress("임무 진입 전 화면 직접 확인")
        else:
            entry_result = self.wait_nav_and_click_until_gone(
                NAV_TEMPLATES["mission"], "홈의 임무 버튼", 0.64,
                timeout=40, verify_delay=5.0, max_clicks=5,
                detect_mission_resume=True,
                next_screen_ready=self._mission_entry_screen_ready,
            )
            if entry_result == "mission_resumed":
                self.log("기존 진행 임무 재개 → 임무 등급 및 관문 진입 단계를 생략합니다.")
                return True
            time.sleep(1.2)
        self.ensure_mission_rank()
        self.select_configured_mission_character_preset()
        self.click_next_stage_if_enabled()
        self.wait_nav_and_click_until_gone(
            NAV_TEMPLATES["mission_enter"], "관문으로 진입", 0.62,
            timeout=40, verify_delay=5.0, max_clicks=5,
            detect_mission_energy_empty=True,
            next_screen_ready=self._mission_combat_screen_ready,
        )
        self.wait_for_mission_energy_popup_after_entry()
        return False
