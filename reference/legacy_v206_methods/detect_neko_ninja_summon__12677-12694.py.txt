    def detect_neko_ninja_summon(self, screen):
        """캐릭터 애니메이션을 제외한 제목·10회 버튼·뽑기 탭 교차 확인."""
        reference, _ = self._cached_template(
            os.path.join(NAV_DIR, "neko_ninja_summon.png"), cv2.IMREAD_COLOR
        )
        if screen is None or reference is None:
            return False, 0.0
        frame = cv2.resize(screen, (480, 1320))
        reference = cv2.resize(reference, (480, 1320))
        scores = []
        for x1, y1, x2, y2 in ((34, 140, 230, 186), (252, 1136, 360, 1190), (399, 1250, 473, 1315)):
            patch = frame[y1:y2, x1:x2]
            template = reference[y1:y2, x1:x2]
            correlation = float(cv2.matchTemplate(patch, template, cv2.TM_CCOEFF_NORMED)[0, 0])
            color_error = float(cv2.absdiff(patch, template).mean())
            scores.append(correlation if color_error <= 25.0 else 0.0)
        score = min(scores)
        return score >= 0.90, score
