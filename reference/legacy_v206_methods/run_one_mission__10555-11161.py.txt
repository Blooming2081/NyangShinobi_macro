    @mission_battle_active
    def run_one_mission(self, resume_in_progress=False):
        """일반 임무 1회 진행 후 success/failed/restarted를 반환합니다."""
        mission_restart_generation = self.restart_generation
        self.pending_mission_skill_tap = None
        self.extreme_wave19_reroll_attempted = False
        self.extreme_reroll_pending_signature = None
        # 새로 입장한 극의 판은 기본 HUD 위치에서 시작합니다. 같은 판의
        # 파란 습득 카드를 선택한 뒤에만 아래 좌표로 전환됩니다.
        self.extreme_new_ninpo_acquired = False
        self.extreme_bigi_allowed_center = BIGI_EXTREME_MISSION_ALLOWED_CENTER
        mission_bigi_center = self._mission_bigi_allowed_center()
        self.last_selected_regular_bigi_stem = None
        # 이전 판의 감시를 먼저 정리합니다. 상위·극의 임무 3번 비기 감시는
        # 1번 카드를 실제 선택했거나 재개 화면의 HUD 얼굴로 선택 상태가
        # 확인된 뒤에만 아래에서 시작합니다.
        self.cancel_mission_special_bigi_monitor()
        self.mission_special_bigi_used_event.clear()
        # 상위·극의 모두 1번 비기 카드를 실제로 선택한 뒤에만 동일한
        # Wave/HP 조건부 허용좌표 감시를 시작합니다.
        resumed_special_bigi_03_available = False
        if resume_in_progress:
            resume_screen = self.adb_screencap()
            if self.mission_rank == "극의 임무":
                self._refresh_extreme_bigi_hud_position(resume_screen)
                mission_bigi_center = self._mission_bigi_allowed_center()
            found_reroll, _, _, _ = self.find_nav_image_near(
                NAV_TEMPLATES["reroll"],
                MISSION_REROLL_CENTER_RATIO,
                MISSION_REROLL_TOLERANCE_RATIO,
                threshold=0.62,
                screen=resume_screen,
            )
            found_layout, layout_score = self.detect_mission_card_layout(
                resume_screen
            )
            acquired_cards = False
            acquired_score = 0.0
            mixed_acquired_cards = False
            mixed_acquired_score = 0.0
            if self.mission_rank == "극의 임무":
                acquired_cards, acquired_score = (
                    self.detect_extreme_three_blue_acquired_cards(resume_screen)
                )
                mixed_acquired_cards, mixed_acquired_score = (
                    self.detect_extreme_mixed_acquired_cards(resume_screen)
                )
            if (
                found_reroll
                or found_layout
                or acquired_cards
                or mixed_acquired_cards
            ):
                if not found_reroll:
                    if acquired_cards:
                        self.log(
                            "재개 직후 다시뽑기 없는 극의 습득 카드 3장 감지 "
                            f"/ 판정값 {acquired_score:.3f} → 즉시 무작위 선택"
                        )
                    elif mixed_acquired_cards:
                        self.log(
                            "재개 직후 다시뽑기 없는 극의 혼합 습득 카드 감지 "
                            f"/ 판정값 {mixed_acquired_score:.3f} → 즉시 우선순위 검사"
                        )
                    else:
                        self.log(
                            "재개 직후 다시뽑기 미감지지만 스킬 카드 화면 감지 "
                            f"/ 판정값 {layout_score:.3f} → 즉시 선택"
                        )
                initial_selection = self.select_mission_skill(
                    screen=resume_screen
                )
            else:
                initial_selection = None
                if (
                    self.bigi_enabled
                    and self.mission_rank in ("상위 임무", "극의 임무")
                ):
                    face_found, _, _, _, face_name = (
                        self.find_bigi_at_allowed_coordinate(
                            mission_bigi_center,
                            threshold=BIGI_MATCH_THRESHOLD,
                            screen=resume_screen,
                            only_deferred=True,
                        )
                    )
                    resumed_special_bigi_03_available = (
                        face_found and self.is_deferred_bigi_name(face_name)
                    )
                self.log("재개된 임무 전투 화면 감시 시작 → 초기 스킬카드 대기를 생략합니다.")
        else:
            initial_selection = self.select_left_mission_skill()
        self.log("일반 임무 진행 감시 시작")

        # 일반 비기와 특수 3번 카드의 선택 상태를 분리합니다. 다른 비기 상태가
        # 먼저 기록돼도 3번 전용 카드 검색은 계속되고, 3번을 실제 선택한 뒤에만
        # 해당 전용 검색을 끝냅니다.
        regular_bigi_card_selected = initial_selection == "bigi"
        special_bigi_03_card_selected = (
            initial_selection == "bigi_deferred"
            or resumed_special_bigi_03_available
        )
        bigi_used_this_battle = False
        roulette_card_guard_until = 0.0
        mission_double_reward_clicked = False
        last_unconfirmed_result_log_at = 0.0
        last_mission_capture_at = 0.0
        last_extreme_monitor_health_check_at = -float("inf")
        if special_bigi_03_card_selected:
            bigi_used_this_battle = True
            if (
                self.bigi_enabled
                and self.mission_rank in ("상위 임무", "극의 임무")
            ):
                self.start_mission_special_bigi_monitor(
                    mission_restart_generation
                )
                if resumed_special_bigi_03_available:
                    self.log(
                        "재개된 상위 임무 HUD에서 3번 비기 얼굴 확인 "
                        "→ 조건부 감시를 시작합니다."
                    )
            if self.mission_rank == "극의 임무":
                self.log(
                    "극의 임무 3번 비기 → 카드만 선택 / Wave 20/20의 "
                    "진행 바 30% 도달 즉시 사용 대기"
                )
            else:
                self.log(
                    "임무 3번 비기 → 카드만 선택 / Wave 20 종료 확정 후 "
                    f"HP {self.mission_special_bigi_hp_threshold()} 이하 전용 사용 대기"
                )
        elif initial_selection == "bigi":
            self.cancel_mission_special_bigi_monitor()
            if self.bigi_enabled:
                bigi_used_this_battle = self.use_selected_bigi_at_allowed_coordinate(
                    "임무",
                    mission_bigi_center,
                    detected_on_card=True,
                )
            else:
                bigi_used_this_battle = True
            if bigi_used_this_battle:
                self.log("임무 비기 사용 확인 → 이번 임무의 비기 검색 종료")
            else:
                self.log("임무 비기 사용 미확인 → 다음 스킬 카드 화면에서 재확인")
        elif (
            self.bigi_enabled
            and initial_selection is not None
            and self.mission_rank not in (
                "하위 임무", "상위 임무", "극의 임무"
            )
        ):
            bigi_used_this_battle = self.use_selected_bigi_at_allowed_coordinate(
                "임무",
                BIGI_MISSION_ALLOWED_CENTER,
                detected_on_card=False,
            )

        while True:
            self.checkpoint()

            # 3번 비기 1번 조건이 확정된 동안에는 HP OCR 전용 스레드가 ADB
            # 캡처를 독점합니다. 5ms 양보만 반복해 2번 조건 감지 간격을 최소화합니다.
            if self.special_bigi_hp_fast_mode.is_set():
                time.sleep(0.005)
                continue

            if self.recover_if_game_crashed("일반 임무 진행 중"):
                return "restarted"

            if self.restart_generation != mission_restart_generation:
                self.log("게임 재시작 확인 → 진행 중이던 임무 상태를 버리고 홈에서 다시 진입합니다.")
                return "restarted"

            mission_capture_interval = MISSION_BATTLE_CAPTURE_INTERVAL_SECONDS
            extreme_bigi_monitor_active = (
                self.mission_rank == "극의 임무"
                and self.special_bigi_monitor_thread is not None
                and self.special_bigi_monitor_thread.is_alive()
            )
            if extreme_bigi_monitor_active:
                # 극의 1번 카드 선택 뒤에는 공유 캡처만 10fps로 공급합니다.
                # 전용 스레드가 별도 캡처를 만들지 않으므로 ADB 중복 부하는 없고,
                # Wave 20 진입을 최장 약 0.1초 안에 검사할 수 있습니다.
                mission_capture_interval = min(
                    mission_capture_interval,
                    EXTREME_BIGI_PRECONDITION_CAPTURE_INTERVAL_SECONDS,
                )
            if (
                self.mission_rank in ("상위 임무", "극의 임무")
                and self.special_bigi_wave20_watch_mode.is_set()
            ):
                # Wave 20 첫 확인 뒤의 교차확인과 진행률 경계는 매우 짧을 수
                # 있습니다. 이 구간만 극의 30ms, 상위 60ms 목표로 높입니다.
                mission_capture_interval = (
                    EXTREME_BIGI_CONFIRM_CAPTURE_INTERVAL_SECONDS
                    if self.mission_rank == "극의 임무"
                    else 0.06
                )
            capture_remaining = (
                last_mission_capture_at
                + mission_capture_interval
                - time.monotonic()
            )
            if capture_remaining > 0.0:
                if self.stop_event.wait(min(0.05, capture_remaining)):
                    return
                continue

            # 한 번의 ADB 캡처를 이번 반복의 모든 일반 판정에 공유합니다.
            # 룰렛 결과 대기와 비기 전용 감시는 화면 전환 속도가 중요하므로
            # 아래의 기존 개별 캡처를 그대로 유지합니다.
            loop_screen = self.adb_screencap()
            last_mission_capture_at = time.monotonic()

            # 날짜·룰렛·카드·결과 판정은 이번 루프의 동일 캡처를 공유합니다.
            if self.handle_date_changed_popup(screen=loop_screen):
                continue

            self.common_recovery_on_frame(loop_screen)

            # ADB input tap 반환만으로 선택이 반영됐다고 취급하지 않습니다.
            # 특히 같은 화면의 카드 배치 검출이 한 프레임 누락돼도 원본
            # 카드 내용이 그대로면 재시도 대기를 해제하지 않습니다.
            if self._verify_pending_mission_skill_tap(loop_screen) == 'waiting':
                continue

            # 극의 1번 카드를 선택한 뒤 다른 카드 처리나 일시적인 감시 스레드
            # 종료로 Wave 20 전제조건 감시가 사라져도 현재 HUD의 3번 얼굴을
            # 근거로 즉시 복구합니다. 별도 ADB 캡처 없이 메인 루프 화면을
            # 재사용하며, 0.5초 간격으로만 확인해 평상시 부하는 제한합니다.
            extreme_health_now = time.monotonic()
            if (
                extreme_health_now
                - last_extreme_monitor_health_check_at >= 0.5
            ):
                last_extreme_monitor_health_check_at = extreme_health_now
                (
                    special_bigi_03_card_selected,
                    monitor_recovered,
                ) = self._ensure_extreme_special_bigi_monitor(
                    loop_screen,
                    special_bigi_03_card_selected,
                    mission_restart_generation,
                )
                if monitor_recovered:
                    bigi_used_this_battle = True

            # 스킬 카드의 다시뽑기/고정 3장 배치를 룰렛보다 먼저 보호
            # 판정합니다. 카드 안의 주황 확인 문양이
            # roulette_ok와 비슷해도 이 프레임에서는 룰렛 클릭을 금지하고
            # 아래의 기존 카드 우선순위 경로로 그대로 넘깁니다.
            card_reroll_found, _, _, card_reroll_val = (
                self.find_nav_image_near(
                    NAV_TEMPLATES["reroll"],
                    MISSION_REROLL_CENTER_RATIO,
                    MISSION_REROLL_TOLERANCE_RATIO,
                    threshold=0.62,
                    screen=loop_screen,
                )
            )
            card_layout_found, card_layout_score = (
                self.detect_mission_card_layout(loop_screen)
            )
            acquired_cards_found = False
            acquired_cards_score = 0.0
            mixed_acquired_cards_found = False
            mixed_acquired_cards_score = 0.0
            if self.mission_rank == "극의 임무" and not card_reroll_found:
                acquired_cards_found, acquired_cards_score = (
                    self.detect_extreme_three_blue_acquired_cards(loop_screen)
                )
                mixed_acquired_cards_found, mixed_acquired_cards_score = (
                    self.detect_extreme_mixed_acquired_cards(loop_screen)
                )
            protected_mission_card_frame = bool(
                card_reroll_found
                or card_layout_found
                or acquired_cards_found
                or mixed_acquired_cards_found
            )

            # 룰렛 확인/스킵은 짧게 나타나므로 다른 전투/스킬 판정보다 먼저 검사합니다.
            found_roulette_ok, roulette_ok_x, roulette_ok_y, roulette_ok_val = (
                self.find_nav_image_near(
                    NAV_TEMPLATES["roulette_ok"],
                    ROULETTE_OK_CENTER_RATIO,
                    ROULETTE_OK_TOLERANCE_RATIO,
                    threshold=0.58,
                    screen=loop_screen,
                )
            )
            if found_roulette_ok and (
                not protected_mission_card_frame
                or roulette_ok_val >= 0.68
            ):
                self.log(
                    f"보스 룰렛 확인 최우선 발견 / 일치율 {roulette_ok_val:.3f} "
                    "→ 즉시 클릭"
                )
                self.adb_click(roulette_ok_x, roulette_ok_y)
                if self.mission_rank in ("상위 임무", "극의 임무"):
                    roulette_card_guard_until = (
                        time.time() + UPPER_MISSION_ROULETTE_CARD_GUARD_SECONDS
                    )
                time.sleep(0.3)
                continue

            found_skip, skip_x, skip_y, skip_val = self.find_nav_image_near(
                NAV_TEMPLATES["roulette_skip"],
                ROULETTE_SKIP_CENTER_RATIO,
                ROULETTE_SKIP_TOLERANCE_RATIO,
                threshold=0.62,
                screen=loop_screen,
            )
            extreme_skip_color_ratio = (
                self._extreme_roulette_skip_color_ratio(loop_screen)
                if self.mission_rank == "극의 임무" and found_skip
                else 0.0
            )
            extreme_skip_confirmed = (
                self.mission_rank != "극의 임무"
                or (
                    skip_val >= EXTREME_ROULETTE_SKIP_CLICK_THRESHOLD
                    and extreme_skip_color_ratio
                    >= EXTREME_ROULETTE_SKIP_PURPLE_RATIO_MIN
                )
            )
            if found_skip and (
                not protected_mission_card_frame or skip_val >= 0.68
            ) and extreme_skip_confirmed:
                self.log(
                    f"보스 룰렛 스킵 최우선 발견 / 일치율 {skip_val:.3f} "
                    + (
                        f"/ 보라색 버튼 {extreme_skip_color_ratio:.3f} "
                        if self.mission_rank == "극의 임무" else ""
                    )
                    + "→ 즉시 클릭"
                )
                self.adb_click(skip_x, skip_y)
                self.log("룰렛 결과 확인 버튼을 기다립니다.")
                roulette_deadline = time.time() + 5.0
                confirmed = False
                while time.time() < roulette_deadline:
                    self.checkpoint()
                    found_ok, ok_x, ok_y, ok_val = self.find_nav_image_near(
                        NAV_TEMPLATES["roulette_ok"],
                        ROULETTE_OK_CENTER_RATIO,
                        ROULETTE_OK_TOLERANCE_RATIO,
                        threshold=0.58,
                    )
                    if found_ok:
                        self.log(f"보스 룰렛 확인 발견 / 일치율 {ok_val:.3f}")
                        self.adb_click(ok_x, ok_y)
                        confirmed = True
                        break
                    time.sleep(0.2)
                if not confirmed:
                    self.log(
                        "룰렛 확인 버튼을 찾지 못함 → 화면이 바뀐 위치를 "
                        "재클릭하지 않고 전투 감시를 계속합니다."
                    )
                if self.mission_rank in ("상위 임무", "극의 임무"):
                    roulette_card_guard_until = (
                        time.time() + UPPER_MISSION_ROULETTE_CARD_GUARD_SECONDS
                    )
                time.sleep(0.3)
                continue

            if (
                self.mission_rank in ("상위 임무", "극의 임무")
                and time.time() < roulette_card_guard_until
            ):
                # 녹화에서 룰렛 확인 뒤 실제 3장 카드 화면까지 약 3.5초의
                # 결과 카드/전환 구간이 확인됐습니다. 이 구간만 건너뜁니다.
                time.sleep(0.05)
                continue

            # 비기 얼굴은 다시뽑기/스킬카드 화면에서만 검사합니다.
            found_reroll = card_reroll_found
            reroll_val = card_reroll_val
            if found_reroll:
                self.log(f"다시뽑기 발견 / 일치율 {reroll_val:.3f} → 스킬 우선순위 검사")
                selection_result = self.select_mission_skill(
                    screen=loop_screen,
                    include_bigi_priority=not regular_bigi_card_selected,
                    include_deferred_bigi_priority=not special_bigi_03_card_selected,
                )
                if selection_result == "bigi_deferred":
                    already_selected_special = special_bigi_03_card_selected
                    special_bigi_03_card_selected = True
                    bigi_used_this_battle = True
                    if (
                        self.bigi_enabled
                        and self.mission_rank in ("상위 임무", "극의 임무")
                        and not (
                            self.mission_rank == "극의 임무"
                            and already_selected_special
                        )
                    ):
                        self.start_mission_special_bigi_monitor(
                            mission_restart_generation
                        )
                    self.log(
                        "임무 3번 비기 → 카드만 선택 / 전용 2단계 조건 대기"
                    )
                elif selection_result == "bigi":
                    regular_bigi_card_selected = True
                    # 극의 1번 비기를 이미 확보했다면 이후 일반 비기 카드
                    # 처리가 Wave 20 전제조건 감시를 취소하면 안 됩니다.
                    if not special_bigi_03_card_selected:
                        self.cancel_mission_special_bigi_monitor()
                    if self.bigi_enabled:
                        bigi_used_this_battle = self.use_selected_bigi_at_allowed_coordinate(
                            "임무",
                            mission_bigi_center,
                            detected_on_card=True,
                        )
                    else:
                        bigi_used_this_battle = True
                    if bigi_used_this_battle:
                        self.log("임무 비기 사용 확인 → 이번 임무의 비기 검색 종료")
                    else:
                        self.log("임무 비기 사용 미확인 → 다음 스킬 카드 화면에서 재확인")
                elif (
                    self.bigi_enabled
                    and not bigi_used_this_battle
                    and self.mission_rank not in (
                        "하위 임무", "상위 임무", "극의 임무"
                    )
                ):
                    bigi_used_this_battle = self.use_selected_bigi_at_allowed_coordinate(
                        "임무",
                        mission_bigi_center,
                        detected_on_card=False,
                    )
                continue

            # 다시 뽑기를 이미 사용해 버튼이 회색으로 바뀐 경우에도
            # 3장 카드의 고정 배치를 확인해 선택 화면을 놓치지 않습니다.
            card_screen = loop_screen
            found_card_layout = (
                card_layout_found
                or acquired_cards_found
                or mixed_acquired_cards_found
            )
            if found_card_layout:
                if acquired_cards_found:
                    self.log(
                        "다시뽑기 없는 극의 습득 카드 3장 화면 감지 "
                        f"/ 판정값 {acquired_cards_score:.3f} → 무작위 선택"
                    )
                elif mixed_acquired_cards_found:
                    self.log(
                        "다시뽑기 없는 극의 혼합 습득 카드 화면 감지 "
                        f"/ 판정값 {mixed_acquired_cards_score:.3f} "
                        "→ 우선순위 검사"
                    )
                else:
                    self.log(
                        "비활성 다시 뽑기 상태의 스킬 카드 화면 감지 "
                        f"/ 판정값 {card_layout_score:.3f} → 스킬 우선순위 검사"
                    )
                selection_result = self.select_mission_skill(
                    screen=card_screen,
                    include_bigi_priority=not regular_bigi_card_selected,
                    include_deferred_bigi_priority=not special_bigi_03_card_selected,
                )
                if selection_result == "bigi_deferred":
                    already_selected_special = special_bigi_03_card_selected
                    special_bigi_03_card_selected = True
                    bigi_used_this_battle = True
                    if (
                        self.bigi_enabled
                        and self.mission_rank in ("상위 임무", "극의 임무")
                        and not (
                            self.mission_rank == "극의 임무"
                            and already_selected_special
                        )
                    ):
                        self.start_mission_special_bigi_monitor(
                            mission_restart_generation
                        )
                    self.log(
                        "임무 3번 비기 → 카드만 선택 / 전용 2단계 조건 대기"
                    )
                elif selection_result == "bigi":
                    regular_bigi_card_selected = True
                    if not special_bigi_03_card_selected:
                        self.cancel_mission_special_bigi_monitor()
                    if self.bigi_enabled:
                        bigi_used_this_battle = self.use_selected_bigi_at_allowed_coordinate(
                            "임무",
                            mission_bigi_center,
                            detected_on_card=True,
                        )
                    else:
                        bigi_used_this_battle = True
                    if bigi_used_this_battle:
                        self.log("임무 비기 사용 확인 → 이번 임무의 비기 검색 종료")
                    else:
                        self.log("임무 비기 사용 미확인 → 다음 스킬 카드 화면에서 재확인")
                elif (
                    self.bigi_enabled
                    and not bigi_used_this_battle
                    and self.mission_rank not in (
                        "하위 임무", "상위 임무", "극의 임무"
                    )
                ):
                    bigi_used_this_battle = self.use_selected_bigi_at_allowed_coordinate(
                        "임무",
                        mission_bigi_center,
                        detected_on_card=False,
                    )
                continue

            found_failed, _, _, failed_val = self.find_nav_image(
                NAV_TEMPLATES["mission_failed"], threshold=0.66,
                screen=loop_screen,
            )
            if found_failed:
                # 완료 화면 처리보다 먼저 별도 HP 감시를 정지시켜, 결과 화면에서
                # OCR 실패 로그나 학습 이미지가 추가로 저장되지 않게 합니다.
                self.cancel_mission_special_bigi_monitor()
                self.capture_learning_event(
                    "mission_failed",
                    f"score={failed_val:.3f}",
                )
                retry_extreme_without_bigi = (
                    self.mission_rank == "극의 임무"
                    and not self.mission_special_bigi_used_event.is_set()
                )
                if retry_extreme_without_bigi:
                    self.log(
                        f"극의 임무 실패 발견 / 일치율 {failed_val:.3f} / "
                        "비기 사용 완료 미확인 → 거울로 전환하지 않고 재도전합니다."
                    )
                else:
                    self.log(
                        f"임무 실패 발견 / 일치율 {failed_val:.3f} "
                        "→ 실패 화면을 닫고 설정된 거울로 전환합니다."
                    )
                # 실패 확인도 성공 확인과 동일한 mission_clear 이미지로
                # 위치를 검출합니다. 화면 닫힘이 확인되지 않으면 다음 분기로
                # 진행하지 않고 동일 임무 결과를 다시 감시합니다.
                if not self.dismiss_mission_failure_confirmation(
                    initial_screen=loop_screen,
                ):
                    continue
                if retry_extreme_without_bigi:
                    return "retry_extreme_without_bigi"
                return "failed"

            # mission_clear는 성공/실패 화면에 공통으로 나타나는 확인 버튼입니다.
            # 확인 버튼만으로 성공을 확정하지 않고, 실패 판정을 먼저 마친 뒤
            # 실제 성공 제목과 확인 버튼이 같은 화면에 있을 때만 성공 처리합니다.
            success_title_found, _, _, success_title_val = self.find_nav_image_near(
                NAV_TEMPLATES["mission_success_title"],
                MISSION_SUCCESS_TITLE_CENTER_RATIO,
                MISSION_SUCCESS_TITLE_TOLERANCE_RATIO,
                threshold=0.76,
                screen=loop_screen,
            )
            found_clear, clear_x, clear_y, clear_val = self.find_nav_image(
                NAV_TEMPLATES["mission_clear"], threshold=0.64,
                screen=loop_screen,
            )
            if success_title_found and found_clear:
                # 완료 로그를 남기기 전에 비기 감시 종료를 확정합니다.
                self.cancel_mission_special_bigi_monitor()
                self.log(
                    "임무 성공 제목 및 확인 버튼 발견 "
                    f"/ 제목 {success_title_val:.3f}, 확인 {clear_val:.3f}"
                )
                confirm_clicked, double_clicked = (
                    self.click_mission_success_confirmation(
                        loop_screen,
                        clear_x,
                        clear_y,
                        "임무 성공 화면",
                        allow_double_reward=not mission_double_reward_clicked,
                    )
                )
                mission_double_reward_clicked = (
                    mission_double_reward_clicked or double_clicked
                )
                if not confirm_clicked:
                    time.sleep(0.25)
                    continue
                time.sleep(1.5)
                self.log("일반 임무 1회 완료")
                self.add_completed_run()
                return "success"
            if found_clear and not success_title_found:
                # 실패 제목도 성공 제목도 검출되지 않은 화면에서는 클릭/완료
                # 기록을 하지 않고, 다음 캡처에서 실패 판정부터 다시 수행합니다.
                now = time.monotonic()
                if now - last_unconfirmed_result_log_at >= 5.0:
                    self.log(
                        "임무 공통 확인 버튼만 발견 / "
                        f"확인 {clear_val:.3f}, 성공 제목 {success_title_val:.3f} "
                        "→ 성공 미확정, 다음 화면에서 성공/실패 재판단"
                    )
                    last_unconfirmed_result_log_at = now

            time.sleep(0.12)
