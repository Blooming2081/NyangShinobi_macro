    def detect_neko_ninja_result(self, screen):
        reference, _ = self._cached_template(
            os.path.join(NAV_DIR, "neko_ninja_result.png"), cv2.IMREAD_COLOR
        )
        if screen is None or reference is None:
            return False
        frame = cv2.resize(screen, (480, 1320))
        reference = cv2.resize(reference, (480, 1320))
        # 어두워진 네코 닌자 제목과 결과의 계속 안내를 모두 확인합니다.
        for x1, y1, x2, y2 in ((34, 140, 230, 186), (170, 970, 312, 994)):
            patch = frame[y1:y2, x1:x2]
            template = reference[y1:y2, x1:x2]
            score = float(cv2.matchTemplate(patch, template, cv2.TM_CCOEFF_NORMED)[0, 0])
            if not score >= 0.90 or float(cv2.absdiff(patch, template).mean()) > 20.0:
                return False
        return True
